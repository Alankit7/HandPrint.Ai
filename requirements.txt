# ── 1. Web framework ──────────────────────────────────────────────────────────
# FastAPI provides the async HTTP layer; uvicorn is the ASGI server.
# python-multipart is required for file upload (Form + UploadFile).
# aiofiles is used by uvicorn for static file serving.
fastapi==0.111.0
uvicorn[standard]==0.29.0
python-multipart==0.0.9
aiofiles==23.2.1


# ── 2. Configuration & validation ─────────────────────────────────────────────
# pydantic v2 for request/response schemas and data validation.
# pydantic-settings reads .env files into the Settings singleton.
# python-dotenv provides the underlying .env parser.
pydantic==2.7.1
pydantic-settings==2.2.1
python-dotenv==1.0.1


# ── 3. Database ───────────────────────────────────────────────────────────────
# SQLAlchemy 2.0 ORM with the new mapped_column / Mapped[] typed API.
# Alembic handles schema migrations (run: alembic upgrade head).
# SQLite is used by default; swap DATABASE_URL in .env for PostgreSQL.
sqlalchemy==2.0.30
alembic==1.13.1


# ── 4. PDF & image I/O ────────────────────────────────────────────────────────
# Pillow: PIL-compatible image library used throughout the ML pipeline.
# PyMuPDF (imported as fitz): rasterises PDF pages to numpy arrays at 300 DPI.
#   Also extracts text blocks, bounding boxes, and embedded images.
# img2pdf: lossless conversion of rendered page images back into a PDF.
#   Preserves DPI metadata; avoids JPEG re-compression artefacts.
# reportlab: assembles debug preview grid PDFs in the preprocessing pipeline.
Pillow==10.3.0
pymupdf==1.24.1
img2pdf==0.5.1
reportlab==4.1.0


# ── 5. Preprocessing pipeline (app/ml/preprocess.py) ─────────────────────────
# numpy: array operations throughout the entire ML pipeline.
# opencv-python-headless: line and word segmentation via morphological ops +
#   connected-component analysis. Headless = no GUI / Qt dependency.
# scipy: deskew correction via horizontal-projection variance maximisation
#   (scipy.ndimage.rotate used at ¼ resolution for speed).
# rapidfuzz: fast fuzzy string alignment (Levenshtein) for matching segmented
#   word crops against a known source transcript (labels.csv alignment).
numpy==1.26.4
opencv-python-headless==4.9.0.80
scipy==1.13.0
rapidfuzz==3.9.3


# ── 6. Scheduling ─────────────────────────────────────────────────────────────
# APScheduler drives the periodic cleanup task (removes stale uploads/outputs).
# BackgroundScheduler is started in app/main.py lifespan and runs in a thread.
apscheduler==3.10.4


# ── 7. Testing (dev / CI only — not installed in production Docker image) ──────
# pytest: test runner (pytest.ini configures testpaths = tests).
# pytest-asyncio: async test support (asyncio_mode = auto in pytest.ini).
# httpx: HTTPX-based test client used by FastAPI's TestClient.
# pytest-mock: provides the mocker fixture for patching in tests.
pytest==8.2.0
pytest-asyncio==0.23.6
httpx==0.27.0
pytest-mock==3.14.0

# ═══════════════════════════════════════════════════════════════════════════════
# OPTIONAL: Kraken line segmenter (soft dependency)
# ═══════════════════════════════════════════════════════════════════════════════
#
# Kraken is a deep-learning OCR and layout analysis toolkit. The preprocessing
# pipeline tries OpenCV segmentation first; if it finds fewer than
# MIN_SEGMENTS_FOR_TRUST lines it falls back to Kraken's blla segmenter.
# If Kraken is not installed the fallback is silently skipped (no error).
#
# Install manually when you want the Kraken fallback:
#   pip install kraken==4.3.13
#
# Note: Kraken pulls in ~500 MB of transitive dependencies (torch, etc.).
# Only install it if the OpenCV segmenter is producing poor results on your
# specific handwriting samples.
